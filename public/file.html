function renderCigaText(value) { if ( value === null || value === undefined || value === "" ) { return ""; } const text = String(value); const tokenRegex = /\[\[([^\]\n|]+)\|(https?:\/\/[^\]\s]+)\]\]/gi; let output = ""; let lastIndex = 0; let match; while ( ( match = tokenRegex.exec(text) ) !== null ) { const fullMatch = match[0]; const label = match[1]; const url = match[2]; const matchStart = match.index; /* Text before the CIGA link. */ output += escapeHTML( text.substring( lastIndex, matchStart ) ); /* Create the real HTML link. */ output += '' + escapeHTML( label ) + ''; lastIndex = matchStart + fullMatch.length; } /* Remaining text after the final link. */ output += escapeHTML( text.substring( lastIndex ) ); /* Preserve line breaks. */ return output.replace( /\r?\n/g, "
" ); }
